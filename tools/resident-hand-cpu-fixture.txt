import * as THREE from 'three';import{it,expect}from'vitest';import{writeFileSync}from'node:fs';import{residentHands as before}from'../../src/presentation/__hand-before';import{residentHands as after}from'../../src/presentation/__hand-after';import{createMeshyNpcRig}from'../../src/presentation/meshynpcs';import{residentMotionLibrary,type WorkContacts}from'../../src/presentation/npc/residentMotion';import{curledVertices,RELAXED_CURL,FIST_CURL}from'../../src/presentation/npc/residentFingers';import type{ResidentBones}from'../../src/presentation/npc/residentRig';import{loadResident,loadResidentMotion,loadResidentRigData}from'./residentFixtures';
it('times paired hand-support preparation on delivered rig geometry',async()=>{const rows=[];for(const id of ['rillford-reeve','ash-recorder']){const{source,entry}=await loadResident(id),data=await loadResidentRigData(id),library=residentMotionLibrary(await loadResidentMotion()),rig=createMeshyNpcRig(source,entry,1,'none',undefined,{}, {data,library,build:'man',seed:7,fighter:false}),resident=rig.resident as unknown as{scene:THREE.Group;bones:ResidentBones;options:{contacts:WorkContacts}},mesh=resident.options.contacts.mesh,plan=rig.npc!.fingers!.plan,positions=mesh.geometry.getAttribute('position').array;
const shapes=(side:'Left'|'Right',vertices:number[])=>[RELAXED_CURL,FIST_CURL].map(curl=>curledVertices(plan,positions,side,curl,vertices));const run=(mode:'before'|'after'|'control')=>{const gen=(mode==='after'?after:before)(mesh,resident.bones,resident.scene,shapes),timings=[];for(;;){const start=performance.now(),r=gen.next();timings.push(performance.now()-start);if(r.done)return{value:r.value,timings};}};
const a=run('before'),b=run('after');expect(b.value).toEqual(a.value);for(let i=0;i<5;i++)for(const mode of ['before','after','control'] as const)run(mode);
for(let i=0;i<24;i++)for(const mode of (i%2?['after','control','before']:['before','control','after']) as ('before'|'after'|'control')[]){const start=performance.now(),r=run(mode);rows.push({id,round:i,mode,ms:performance.now()-start,maxResumeMs:Math.max(...r.timings),turns:r.timings.length});}}
writeFileSync(process.env.HAND_CPU_OUT!,JSON.stringify({rows}));},60000);
